cmake_minimum_required(VERSION 3.16)
project(macroui-probes C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(CMAKE_CONFIGURATION_TYPES)
  foreach(cfg ${CMAKE_CONFIGURATION_TYPES})
    string(TOUPPER "${cfg}" cfg_upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg_upper} "${CMAKE_BINARY_DIR}")
  endforeach()
else()
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()

add_executable(slow-math-test slow_math.c slow_math_test.c)
if(MSVC)
  target_compile_options(slow-math-test PRIVATE /W4)
else()
  target_compile_options(slow-math-test PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  add_executable(move-probe move_probe.c)
  target_link_libraries(move-probe PRIVATE user32 winmm)

  add_executable(slow-probe slow_probe.c slow_math.c)
  target_link_libraries(slow-probe PRIVATE user32)
  if(MSVC)
    target_compile_options(slow-probe PRIVATE /W4)
  else()
    target_compile_options(slow-probe PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
